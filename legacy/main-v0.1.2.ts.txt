function bind() {
  document
    .querySelectorAll<HTMLElement>("[data-go]")
    .forEach((e) => (e.onclick = () => go(e.dataset.go!)));
  document.querySelectorAll<HTMLElement>("[data-mode]").forEach(
    (e) =>
      (e.onclick = () => {
        mode = e.dataset.mode as Mode;
        render();
      }),
  );
  document.querySelectorAll<HTMLElement>("[data-input]").forEach(
    (e) =>
      (e.onclick = () => {
        input = e.dataset.input as Input;
        render();
      }),
  );
  $("#begin")?.addEventListener("click", enableCamera);
  $("#play")?.addEventListener("click", startGame);
  $("#pause")?.addEventListener("click", () => pause("稍作休息，准备好后继续"));
  $("#reverse")?.addEventListener("change", (e) => {
    reverse = (e.target as HTMLInputElement).checked;
  });
  $("#duration")?.addEventListener("click", () => {
    duration = duration === 60 ? 32 : 60;
    render();
  });
  $("#low")?.addEventListener("change", (e) => {
    low = (e.target as HTMLInputElement).checked;
    stage.low = low;
  });
  $("#preview")?.addEventListener("change", (e) => {
    preview = (e.target as HTMLInputElement).checked;
    camera.setPreviewVisible(preview);
  });
  $("#how-to")?.addEventListener("click", () => showTutorial(false));
  $("#export")?.addEventListener("click", exportReport);
  $("#sync")?.addEventListener("click", startSync);
  $("#clear")?.addEventListener("click", () => {
    if (confirm("清除本站的校准、成绩和设置？已导出文件不受影响。")) {
      for (const k of [
        "pulse-calibration",
        "pulse-offset",
        "pulse-bests",
        "pulse-tutorial",
      ])
        localStorage.removeItem(k);
      calibration = null;
      tracker.reset(null);
      sessionVerified = false;
      offset = 0;
      trials = [];
      logs = [];
      tutorialSeen = false;
      render();
      toast("本地档案已清除");
    }
  });
  $("#threshold")?.addEventListener("input", (e) => {
    if (calibration) {
      calibration.threshold = Number((e.target as HTMLInputElement).value);
      tracker.reset(calibration);
      save("pulse-calibration", calibration);
      log({
        type: "parameter",
        threshold: calibration.threshold,
        time: Date.now(),
      });
      $("#threshold-label").textContent =
        (calibration.threshold * 100).toFixed(1) + "%";
    }
  });
  $("#reset-origin")?.addEventListener("click", () => {
    if (!frame?.quality) return toast("先让双肩清晰入镜");
    const c = makeCalibration(
      Array(25).fill(frame.points),
      calibration?.profile ?? "full-body",
    );
    if (c) {
      c.z0 = calibration?.z0 ?? false;
      c.z1 = calibration?.z1 ?? false;
      c.threshold = calibration?.threshold ?? 0.035;
      calibration = c;
      tracker.reset(c);
      save("pulse-calibration", c);
      log({ type: "origin-reset", time: Date.now() });
      toast("已明确重设参考站位");
    }
  });
  $("#trial-start")?.addEventListener("click", () => {
    if (!camera.active || !sessionVerified)
      return toast("先开启摄像头并完成站位校准");
    if (calibration?.profile === "upper-body")
      return toast("当前为上半身XY模式；Z动作试验需要重新完成双肩双髋校准");
    if (trial) return;
    trial = {
      label: $("#trial-label")
        ? ($("#trial-label") as HTMLSelectElement).value
        : "still",
      start: performance.now(),
      events: [],
      auto: false,
    };
  });
}

function go(next: string) {
  if (playing || counting) {
    music.pause();
    playing = false;
    counting = false;
    paused = false;
  }
  guide = null;
  $("#guide")?.remove();
  $("#tutorial")?.remove();
  trial = null;
  stopSync();
  view = next;
  stage.active = false;
  render();
}

    camera.mountPreview(slot, preview);
  }
}
function updateDevice() {
  const active = camera.active;
  $("#camera-dot").className = active ? "live" : "";
  $("#camera-status").textContent = active
    ? camera.health.message
    : "摄像头未开启";
  $("#camera-power").hidden = !active;
}
async function enableCamera() {
  if (cameraLoading) return;
  cameraLoading = true;
  const begin = $("#begin") as HTMLButtonElement | null;
  if (begin) {
    begin.disabled = true;
    begin.textContent = "准备摄像头…";
  }
  try {
    if (!camera.active) {
      toast("正在准备摄像头与本地识别模型…");
      await camera.start();
    }
    input = "camera";
    sessionVerified = false;
    view = "lab";
    render();
    beginGuide();
  } catch (e) {
    toast(`无法开启摄像头：${String(e)}。请检查浏览器权限与其他相机应用。`);
    log({ type: "camera-error", error: String(e) });
    render();
  } finally {
    cameraLoading = false;
    const b = $("#begin") as HTMLButtonElement | null;
    if (b) {
      b.disabled = false;
      b.textContent = camera.active ? "站位与 Z 校准 ↗" : "开启摄像头 ↗";
    }
  }
}

function finishGuide() {
  if (calibration) {
    calibration.z0 =
      calibration.profile !== "upper-body" && (guide?.depth ?? 0) >= 3;
    calibration.z1 =
      calibration.profile !== "upper-body" && (guide?.correct ?? 0) >= 4;
    save("pulse-calibration", calibration);
    tracker.reset(calibration);
    sessionVerified = true;
    log({
      type: "calibration",
      data: { ...calibration },
      z0Count: guide?.depth,
      z1Correct: guide?.correct,
      z1Answered: guide?.answered,
      z1Total: guide?.total,
    });
    toast(
      calibration.profile === "upper-body"
        ? "上半身校准已保存，可以先玩 XY；Z 未启用"
        : `站位已保存 · Z0 ${calibration.z0 ? "通过" : "待验证"} · Z1 ${calibration.z1 ? "通过" : "待验证"}`,
    );
  }
  guide = null;
  $("#guide")?.remove();
  render();
}
function advanceGuide(now: number) {
  if (!guide) return;
  const g = guide,
    elapsed = (now - g.start) / 1000;
  if (g.step === 0) {
    const points = frame?.points ?? [],
      diagnosis = inspectPose(points);
    const c = capture.candidate("full-body", now);
    $("#guide-joints").innerHTML = jointPanel(points);
    $("#guide-state").textContent = diagnosis.fullReady
      ? "四点可见，保持站稳"
      : diagnosis.shoulderReady
        ? "双肩可见，髋部尚未通过"
        : "等待双肩清晰入镜";
    $("#guide-progress").style.width =
      capture.progress("full-body", now) * 100 + "%";
    $("#guide-detail").textContent = diagnosis.fullReady
      ? "保持稳定约2秒；Z校准不需要膝盖或脚踝点。"
      : diagnosis.reason +
        "。可后退一点或调整镜头俯仰，让腰带以下/上大腿也入镜；或选择上半身模式先玩XY。";
    const upperButton = $("#upper-body") as HTMLButtonElement | null;
    if (upperButton)
      upperButton.disabled = !capture.candidate("upper-body", now);
    if (c) {
      calibration = c;
      tracker.reset(c);
      g.step = 1;
      g.start = now;
      showGuide();
    }
  } else if (g.step === 1) {
    $("#guide-state").textContent =
      `舒适伸展 ${Math.max(0, 6 - Math.floor(elapsed))}`;
    $("#guide-progress").style.width = Math.min(100, (elapsed / 6) * 100) + "%";
    if (elapsed >= 6) {
      if (calibration && g.left.length > 15) {
        const xs = g.left,
          ys = g.right;
        const span = Math.max(...xs) - Math.min(...xs),
          vspan = Math.max(...ys) - Math.min(...ys);
        calibration.reachX = Math.max(calibration.shoulder * 1.9, span * 0.85);
        calibration.reachY = Math.max(
          calibration.profile === "upper-body"
            ? calibration.shoulder * 2.5
            : calibration.torso * 2.1,
          vspan * 1.2,
        );
        tracker.reset(calibration);
      }
      if (calibration?.profile === "upper-body") {
        finishGuide();
        return;
      }
      g.step = 2;
      g.start = now;
      trial = null;
      showGuide();
    }
  } else {
    const rounds = g.step === 2 ? 3 : 4,
      round = Math.floor(elapsed / 5);
    if (round >= rounds) {
      if (trial) completeTrial(true);
      if (g.step === 2) {
        g.step = 3;
        g.start = now;
        trial = null;
        tracker.reset(calibration);
        showGuide();
      } else finishGuide();
      return;
    }
    const phase = elapsed % 5;
    const label =
      g.step === 2
        ? "either"
        : round % 2 === 0
          ? "toward_camera"
          : "away_from_camera";
    const expectedStart = g.start + round * 5000;
    if (trial && trial.start !== expectedStart) completeTrial(true);
    if (!trial) {
      trial = { label, start: expectedStart, events: [], auto: true };
    }
    const instruction =
      phase < 1
        ? "准备"
        : phase < 3
          ? label === "either"
            ? "身体前后动一下"
            : label === "toward_camera"
              ? "靠近摄像头"
              : "远离摄像头"
          : "轻松回位，停稳";
    $("#guide-state").textContent = `${round + 1}/${rounds} · ${instruction}`;
    $("#guide-progress").style.width = (elapsed / (rounds * 5)) * 100 + "%";
    $("#guide-detail").textContent =
      `Z0 合格事件 ${g.depth} / 3 · Z1 方向正确 ${g.correct} / 4（校准结果须独立验证）`;
  }
}
function completeTrial(auto: boolean) {
  if (!trial) return;
  const t = trial;
  const e = t.events[0];
  const prediction = e?.direction ?? (e ? "uncertain" : "no_event");
  if (auto && guide) {
    if (guide.step === 2 && e) guide.depth++;
    if (guide.step === 3) {
      guide.total++;
      if (e?.direction) guide.answered++;
      if (prediction === t.label) guide.correct++;
    }
  }
  const tester =
    ($("#tester") as HTMLInputElement | null)?.value ?? "calibration";
  trials.push({
    label: t.label,
    prediction,
    confidence: e?.confidence ?? 0,
    phase: "movement-window",
    time: Date.now(),
    tester,
    verifiedByHuman: !auto,
  });
  log({
    type: auto ? "calibration-trial" : "trial",
    label: t.label,
    prediction,
    events: t.events.map((e) => ({
      id: e.id,
      direction: e.direction,
      confidence: e.confidence,
    })),
    verifiedByHuman: !auto,
  });
  trial = null;
  $("#trial-count")?.replaceChildren(
    document.createTextNode(`已记录 ${trials.length} 条试验`),
  );
}

async function startGame() {
  if (input === "camera") {
    if (!camera.active) {
      await enableCamera();
      return;
    }
    if (!sessionVerified) {
      beginGuide();
      view = "lab";
      render();
      return;
    }
    if (
      (mode === "Z0" && !calibration?.z0) ||
      (mode === "Z1" && !calibration?.z1)
    )
      return toast("此档 Z 校准尚未通过。请在实验室重试，或选择较低配置。");
  }
  if (!tutorialSeen) {
    showTutorial(true);
    return;
  }
  try {
    const r = await fetch(
      `/charts/${duration}-${mode}${reverse && mode === "Z1" ? "-reverse" : ""}.json`,
    );
    if (!r.ok) throw Error("谱面读取失败");
    const chart: Chart = await r.json();
    const errors = validateChart(chart);
    if (errors.length) throw Error(errors.join(" / "));
    await music.init();
    game = new Game(chart);
    stage.game = game;
    stage.frame = input === "camera" ? frame : null;
    stats = { processing: [], fps: [], judgements: [], interruptions: [] };
    lastZ = -Infinity;
    tracker.reset(calibration);
    game.onJudge = (j) => {
      stage.flash(j);
      music.hit(j.grade);
      stats.judgements.push({
        id: j.note.id,
        time: j.note.time,
        grade: j.grade,
        error: j.error,
        action: j.action,
      });
    };
    view = "play";
    playing = true;
    paused = false;
    counting = true;
    finishAt = 0;
    lastPose = performance.now();
    badSince = 0;
    lowSince = 0;
    render();
    music.play(0, 3);
    log({
      type: "game-start",
      chart: chart.id,
      input,
      mode,
      rule: RULE_VERSION,
      time: Date.now(),
    });
  } catch (e) {
    toast(`无法开始：${String(e)}`);
  }
}

  if (!playing || paused) return;
  music.pause();
  playing = false;
  paused = true;
  counting = false;
  if (game) game.assisted = true;
  stats.interruptions.push(reason);
  pauseReason = reason;
  const el = $("#pause-overlay");
  if (!el) return;
  el.hidden = false;
  el.className = "modal-backdrop";
  el.innerHTML = `<div class="modal"><div class="eyebrow">STAGE PAUSED</div><h2>舞台等你回来。</h2><p>${escape(pauseReason)}</p><p class="fineprint">本局标记为辅助成绩。继续将保留已完成判定，过期动作会被丢弃。</p><div class="modal-actions"><button class="text-btn" data-go="home">返回</button><button class="secondary" id="restart">重新开始</button><button class="primary" id="resume">继续 ▷</button></div></div>`;
  el.querySelector("[data-go]")!.addEventListener("click", () => go("home"));
  $("#restart").onclick = startGame;
  $("#resume").onclick = async () => {
    if (
      input === "camera" &&
      (!(mode === "XY" ? frame?.quality : frame?.depthQuality) ||

    )
      return toast(

function finish() {
  if (!game) return;
  music.pause();
  playing = false;
  counting = false;
  paused = false;
  const bests = safe<Record<string, number>>("pulse-bests", {});
  if (!game.assisted) {
    bests[bestKey()] = Math.max(bests[bestKey()] ?? 0, game.score);
    save("pulse-bests", bests);
  }
  log({
    type: "game-end",
    score: game.score,
    counts: game.counts,
    assisted: game.assisted,
  });
  view = "results";
  render();
}

function dispatch(a: Action) {
  if (!playing || !game || music.time < 0) return;
  const time = music.eventTime(a.time) - offset / 1000;
  if (a.kind === "depth") {
    if (time < lastZ + 1.5) return;
    lastZ = time;
    if (
      mode === "Z1" &&
      (!a.direction || (a.directionConfidence ?? 0) < 0.75)
    ) {
      game.protect(time - 0.38, time + 0.38);
      return;
    }
  }
  game.consume({ ...a, time });
}

camera.onPose = (points, time, processing) => {
  lastPose = performance.now();
  if (guide?.step === 0) capture.add(points, time);
  if (guide?.step === 1 && points.length > 16) {
    for (const i of [15, 16])
      if ((points[i].visibility ?? 0) > 0.65) {
        guide.left.push(points[i].x);
        guide.right.push(points[i].y);
      }
  }
  frame = tracker.process(points, time, processing);
  if (input === "camera") stage.frame = frame;
  if (playing && mode === "Z1" && input === "camera" && frame.reason && game) {
    const at = music.eventTime(time) - offset / 1000;
    const n = game.chart.notes.find(
      (n) =>
        n.kind === "step" &&
        !game!.results.has(n.id) &&
        Math.abs(n.time - at) <= 0.38,
    );
    if (n) game.protect(n.time - 0.001, n.time + 0.001);
  }
  if (mode === "XY" ? frame.quality : frame.depthQuality) {
    badSince = 0;
  } else if (!badSince) badSince = performance.now();
  if (
    playing &&
    input === "camera" &&
    badSince &&
    performance.now() - badSince > 300
  ) {
    game?.protect(music.time - 0.3, music.time + 0.15);
    pause(
      mode === "XY"
        ? "双肩识别暂时丢失，请回到取景范围"
        : "Z需要双肩和双髋，请查看实验室具体缺失点",
    );
  }
  for (const a of frame.actions) {
    if (a.kind === "depth") {
      log({ type: "depth", ...a, state: frame.status, scale: frame.scale });
      if (trial) {
        const phase = (a.time - trial.start) / 1000;
        if (phase >= 1 && phase < 3) trial.events.push(a);
      }
    }
    dispatch(a);
  }
  if (playing) {
    stats.processing.push(processing);
    stats.fps.push(camera.fps);
  }
};
function keyAction(key: string) {
  if (!playing || !game || music.time < 0) return;
  const now = performance.now(),
    time = music.time - offset / 1000;
  if (key === "w" || key === "s") {
    dispatch({
      id: `key-${currentKey++}`,
      time: now,
      kind: "depth",
      direction: key === "w" ? "toward_camera" : "away_from_camera",
      directionConfidence: 1,
      confidence: 1,
    });
    return;
  }
  if (key === "a" || key === "d") {
    dispatch({
      id: `key-${currentKey++}`,
      time: now,
      kind: "sway",
      confidence: 1,
    });
    return;
  }
  const hand = key === "j" || key === "arrowup" ? "left" : "right";
  const targets = game.chart.notes
    .filter(
      (n) =>
        !game!.results.has(n.id) &&
        (!n.hand || n.hand === hand) &&
        ["catch", "slash", "double"].includes(n.kind),
    )
    .sort((a, b) => Math.abs(a.time - time) - Math.abs(b.time - time));
  const n = targets[0];
  if (!n || Math.abs(n.time - time) > 0.5) return;
  const x = n.kind === "double" ? (hand === "left" ? 0.32 : 0.68) : n.x,
    y = n.y;
  const slash = key.startsWith("arrow");
  const dy = key === "arrowup" ? -1 : 1;
  stage.simulationHands[hand] = { x, y };
  dispatch({
    id: `key-${currentKey++}`,
    time: now,
    kind: "hand",
    hand,
    x,
    y,
    prevX: x,
    prevY: slash ? y - dy * 0.15 : y - 0.09,
    dx: 0,
    dy: slash ? dy : 0,
    speed: slash ? 2 : 0.5,
    confidence: 1,
  });
}

  if (sync) return;

    document.body.append(el);
    $("#tap").onclick = recordTap;
    $("#sync-cancel").onclick = stopSync;
    sync!.timer = window.setInterval(() => {
      if (!sync) return;
      const now = performance.now();
      if (now >= sync.nextBeat) {
        music.hit("Perfect");
        sync.beats++;
        sync.nextBeat += 500;
        $("#tap").classList.toggle("beat");
        if (sync.beats > 24) stopSync();
      }
    }, 12);
  });

function recordTap() {
  if (!sync || performance.now() < sync.start) return;
  const delta = performance.now() - sync.start;
  const residual = delta - Math.round(delta / 500) * 500;
  sync.taps.push(residual);
  $("#tap-count").textContent = `${sync.taps.length}/8`;
  if (sync.taps.length >= 8) {
    const values = sync.taps.slice(1).sort((a, b) => a - b);
    offset = Math.round(values[Math.floor(values.length / 2)]);
    save("pulse-offset", offset);
    log({ type: "sync", offset, taps: sync.taps });
    stopSync();
    render();
    toast(`综合偏移已保存 ${offset}ms；此为人工敲拍估计，非端到端延迟实测。`);
  }
}
function stopSync() {
  if (sync) clearInterval(sync.timer);
  sync = null;
  $("#sync-modal")?.remove();
}
function exportReport() {
  const rows = trials;
  const directionRows = rows.filter(
    (r) =>
      r.verifiedByHuman &&
      (r.label === "toward_camera" || r.label === "away_from_camera"),
  );
  const answered = directionRows.filter(
    (r) =>
      r.prediction === "toward_camera" || r.prediction === "away_from_camera",
  );
  const correct = directionRows.filter((r) => r.label === r.prediction).length;
  const pct = (n: number, d: number) => (d ? n / d : null);
  const percentile = (a: number[], p: number) =>
    a.length
      ? [...a].sort((x, y) => x - y)[
          Math.min(a.length - 1, Math.floor(a.length * p))
        ]
      : null;
  const report = {
    app: "Pulse Stage",
    version: "0.1.2",
    createdAt: new Date().toISOString(),
    userAgent: navigator.userAgent,
    rule: RULE_VERSION,
    input,
    mode,
    calibration,
    poseDiagnostics: frame
      ? {
          modelOutputCount: frame.points.length,
          joints: inspectPose(frame.points).joints,
          reason: inspectPose(frame.points).reason,
        }
      : null,
    offsetMs: offset,
    privacy: "no raw video or continuous landmarks included",
    model: POSE_MODEL,
    inputHealth: camera.health,
    verification:
      "Prototype only. Human confirmation and independent validation required. Keyboard runs are not camera evidence.",
    chart: game?.chart,
    game: game
      ? { score: game.score, counts: game.counts, assisted: game.assisted }
      : null,
    metrics: {
      directionTrials: directionRows.length,
      wholeSampleSuccess: pct(correct, directionRows.length),
      coverage: pct(answered.length, directionRows.length),
      answeredAccuracy: pct(correct, answered.length),
      softwareInferenceP95Ms: percentile(stats.processing, 0.95),
      poseFpsMedian: percentile(stats.fps, 0.5),
      endToEndLatencyMs: null,
    },
    trials,
    logs,
    stats,
  };
  const url = URL.createObjectURL(
    new Blob([JSON.stringify(report, null, 2)], { type: "application/json" }),
  );
  const a = document.createElement("a");
  a.href = url;
  a.download = `pulse-stage-report-${Date.now()}.json`;
  a.click();
  setTimeout(() => URL.revokeObjectURL(url), 2000);
  toast("已导出本机报告，不含原始视频或连续骨架轨迹");
}

      missing: "无数据",
    };
  return `<div class="joint-summary">模型输出 ${points.length}/33 点 · 青色可靠 / 琥珀色为待确认估计 · 校准阈值 ${Math.round(CALIBRATION_CONFIDENCE * 100)}%</div><div class="joint-grid">${d.joints.map((j) => `<div class="joint-item ${j.status}"><b>${j.label}</b><span>${label[j.status]} · ${Math.round(Math.min(j.visibility, j.presence ?? 1) * 100)}%</span></div>`).join("")}</div>`;
}
function labPaint() {
  if (view !== "lab") return;
  const c = $<HTMLCanvasElement>("#skeleton");
  if (c) {
    const r = c.getBoundingClientRect();
    c.width = r.width * devicePixelRatio;
    c.height = r.height * devicePixelRatio;
    const ctx = c.getContext("2d")!;
    ctx.scale(devicePixelRatio, devicePixelRatio);
    const scale = Math.min(
      r.width / (camera.video.videoWidth || 640),
      r.height / (camera.video.videoHeight || 480),
    );
    const vw = (camera.video.videoWidth || 640) * scale,
      vh = (camera.video.videoHeight || 480) * scale;
    ctx.translate((r.width - vw) / 2, (r.height - vh) / 2);
    if (frame?.points.length) {
      const p = frame.points;
      ctx.strokeStyle = "#69e9ff";
      ctx.lineWidth = 2;
      for (const [a, b] of [
        [11, 12],
        [11, 13],
        [13, 15],
        [12, 14],
        [14, 16],
        [11, 23],
        [12, 24],
        [23, 24],
        [23, 25],
        [25, 27],
        [24, 26],
        [26, 28],
      ]) {
        if (
          !p[a] ||
          !p[b] ||
          [p[a], p[b]].some(
            (point) => !Number.isFinite(point.x) || !Number.isFinite(point.y),
          )
        )
          continue;
        const certain =
          jointStatus(p[a]) === "ready" && jointStatus(p[b]) === "ready";
        ctx.strokeStyle = certain ? "#69e9ff" : "#e5b67a88";
        ctx.setLineDash(certain ? [] : [4, 4]);
        ctx.beginPath();
        ctx.moveTo((1 - p[a].x) * vw, p[a].y * vh);
        ctx.lineTo((1 - p[b].x) * vw, p[b].y * vh);
        ctx.stroke();
      }
      ctx.setLineDash([]);
      for (let i = 0; i < p.length; i++) {
        if (
          !Number.isFinite(p[i].x) ||
          !Number.isFinite(p[i].y) ||
          p[i].x < 0 ||
          p[i].x > 1 ||
          p[i].y < 0 ||
          p[i].y > 1
        )
          continue;
        const status = jointStatus(p[i]);
        ctx.fillStyle =
          status === "ready" ? (i === 16 ? "#ff88c8" : "#69e9ff") : "#e5b67a88";
        ctx.beginPath();
        ctx.arc(
          (1 - p[i].x) * vw,
          p[i].y * vh,
          [11, 12, 23, 24].includes(i) ? 5 : 2.5,
          0,
          Math.PI * 2,
        );
        ctx.fill();
      }
      ctx.font = "12px system-ui";
      ctx.fillStyle = "#fff";
      for (const j of inspectPose(p).joints) {
        const point = p[j.id];
        if (point && Number.isFinite(point.x) && Number.isFinite(point.y))
          ctx.fillText(
            j.label,
            (1 - Math.max(0.03, Math.min(0.97, point.x))) * vw + 7,
            Math.max(0.04, Math.min(0.96, point.y)) * vh - 7,
          );
      }
    }
  }
  $("#joint-diagnostics").innerHTML = jointPanel(frame?.points ?? []);
  if (!frame) return;
  $("#lab-quality").textContent =
    `${frame.points.length} 个模型输出点 · ${inspectPose(frame.points).reason}`;
  $("#direction-label").textContent =
    frame.direction === "toward_camera"
      ? "靠近 ↘"
      : frame.direction === "away_from_camera"
        ? "远离 ↗"
        : frame.direction === "uncertain"
          ? "不确定"
          : "稳定 / 等待动作";
  $("#signal-reason").textContent = frame.reason || frame.status;
  $("#scale-value").textContent = frame.scale.toFixed(3) + "×";
  $("#confidence-value").textContent = Math.round(frame.confidence * 100) + "%";
  $("#inference-value").textContent = frame.processing.toFixed(0) + " ms";
  history.push(frame.scale);
  if (history.length > 120) history.shift();
  const graph = $<HTMLCanvasElement>("#signal-graph");
  if (graph) {
    const r = graph.getBoundingClientRect();
    graph.width = r.width * devicePixelRatio;
    graph.height = r.height * devicePixelRatio;
    const ctx = graph.getContext("2d")!;
    ctx.scale(devicePixelRatio, devicePixelRatio);
    ctx.strokeStyle = "#ffffff18";
    for (let i = 1; i < 4; i++) {
      ctx.beginPath();
      ctx.moveTo(0, (r.height * i) / 4);
      ctx.lineTo(r.width, (r.height * i) / 4);
      ctx.stroke();
    }
    ctx.strokeStyle = "#63e6ff";
    ctx.lineWidth = 2;
    ctx.beginPath();
    history.forEach((v, i) => {
      const x = (i / 119) * r.width,
        y = r.height * 0.5 - (v - 1) * r.height * 2;
      if (i) ctx.lineTo(x, y);
      else ctx.moveTo(x, y);
    });
    ctx.stroke();
  }
}
function dwell(now: number) {
  if (
    !camera.active ||
    !frame?.quality ||
    performance.now() - lastPose > 1500 ||
    playing ||
    guide
  )
    return;
  const p = frame.hands.right ?? frame.hands.left;
  if (!p) return;
  const cursor = $("#hand-cursor");
  cursor.classList.add("visible");
  const x = p.x * innerWidth,
    y = p.y * innerHeight;
  cursor.style.left = x + "px";
  cursor.style.top = y + "px";
  const target =
    document
      .elementFromPoint(x, y)
      ?.closest<HTMLElement>("button:not(:disabled)") ?? null;
  if (target !== navDwell.el) {
    navDwell = { el: target, since: now };
    cursor.style.setProperty("--progress", "0deg");
  } else if (target) {
    const progress = (now - navDwell.since) / 1000;
    cursor.style.setProperty(
      "--progress",
      Math.min(360, progress * 360) + "deg",
    );
    if (progress >= 1) {
      navDwell = { el: null, since: now };
      target.click();
    }
  }
}
function animate(now: number) {
  requestAnimationFrame(animate);
  stage.time = music.time;
  stage.draw(now);
  if (guide) advanceGuide(now);
  if (trial && !trial.auto) {
    const elapsed = (now - trial.start) / 1000;
    const state =
      elapsed < 1 ? "准备" : elapsed < 3 ? "请完成所选动作" : "回位并停稳";
    $("#trial-status")?.replaceChildren(
      document.createTextNode(
        `${state} · ${Math.max(0, 5 - elapsed).toFixed(1)} 秒`,
      ),
    );
    if (elapsed >= 5) {
      const valid = confirm("本次是否按提示完成动作？确认保存，取消丢弃。");
      if (valid) completeTrial(false);
      else trial = null;
      $("#trial-status")?.replaceChildren(
        document.createTextNode(
          "试验结束，可在导出报告中检查预测与真实动作标签。",
        ),
      );
    }
  }
  if (view === "play" && playing && game) {
    const time = music.time;
    const msg = $("#center-message");
    if (counting) {
      if (time < 0)
        msg.innerHTML = `<strong>${Math.ceil(-time)}</strong><span>跟随节拍，准备发光</span>`;
      else {
        counting = false;
        msg.innerHTML = "";
      }
    }
    game.tick(time);
    $("#score").textContent = String(game.score).padStart(7, "0");
    $("#combo").textContent = String(game.combo);
    $("#time").textContent =
      `${Math.floor(Math.max(0, time) / 60)}:${String(Math.floor(Math.max(0, time)) % 60).padStart(2, "0")}`;
    $("#progress").style.width = Math.max(0, (time / duration) * 100) + "%";
    const next = game.chart.notes.find(
      (n) =>
        ["pulse", "step", "double"].includes(n.kind) &&
        !game!.results.has(n.id) &&
        n.time > time &&
        n.time - time < 2 &&
        n.id.startsWith("z"),
    );
    $("#z-cue").innerHTML = next
      ? `<small>空间节点 / ${Math.ceil((next.time - time) * 2)} 拍</small><strong>${next.kind === "double" ? "双手接拍" : next.kind === "pulse" ? "前后动一下" : next.direction === "toward_camera" ? "靠近摄像头" : "远离摄像头"}</strong>`
      : "";
    if (input === "camera") {
      $("#tracking-status").textContent =
        frame?.reason ||
        `${frame?.status ?? "等待输入"} · ${camera.fps.toFixed(0)} FPS`;
      const health = camera.health;
      if (health.pause) pause(health.message);
      if (!counting && health.slow) {
        if (!lowSince) lowSince = now;
        if (now - lowSince > 2000) {
          if (!game.assisted) {
            toast("识别偏慢，已降低特效；本局继续练习并标记为辅助成绩");
            stats.interruptions.push("slow-inference-practice");
          }
          game.assisted = true;
          stage.low = true;
          $("#tracking-status").textContent = health.message;
        }
      } else lowSince = 0;
      const p = frame?.points;
      if (
        p &&
        p.length > 16 &&
        p[15].y < p[0].y &&
        p[16].y < p[0].y &&
        Math.abs(p[15].x - p[16].x) < 0.22
      ) {
        if (!aboveSince) aboveSince = now;
        if (now - aboveSince > 1500) {
          pause("举手暂停");
          aboveSince = 0;
        }
      } else aboveSince = 0;
    }
    if (time >= duration + 0.75) {
      if (!finishAt) finishAt = now;
      finish();
    }
  }
  if (now - lastPaint > 100) {
    lastPaint = now;
    labPaint();
    updateDevice();
    $("#fps").textContent =
      `${stage.fps.toFixed(0)} RENDER FPS${camera.active ? " / " + camera.fps.toFixed(0) + " POSE FPS" : ""}`;
  }
  if (!playing) dwell(now);
  else $("#hand-cursor").classList.remove("visible");
}
render();
requestAnimationFrame(animate);
window.addEventListener("pagehide", () => {
  camera.stop();
  music.pause();
});
