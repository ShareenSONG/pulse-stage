function guideTick(now: number) {
  if (!guide) return;
  const g = guide,
    d = inspectUpper(frame?.points ?? []),
    fresh = !!frame && now - frame.time < 250;
  let progress = 0,
    detail = d.reason;
  if (g.step === 0) {
    const c = calibrateUpper(capture.samples);
    progress = capture.progress("upper-body", now);
    if (c && fresh) {
      calibration = c;
      tracker.reset(c);
      g.step = 1;
      g.start = now;
      showGuide();
      return;
    }
  } else if (g.step === 1 || g.step === 2) {
    const i = g.step === 1 ? 15 : 16,
      s = g.step === 1 ? 11 : 12;
    const ready =
      fresh && d.ready && frame!.points[i].y < frame!.points[s].y - 0.06;
    if (ready) {
      if (!g.held) g.held = now;
      progress = (now - g.held) / 500;
    } else g.held = 0;
    if (progress >= 1) {
      g.step++;
      g.start = now;
      g.held = 0;
      showGuide();
      return;
    }
  } else {
    progress = Math.min(1, (now - g.start) / 8000);
    detail = `舒适伸展 ${Math.max(0, Math.ceil(8 - (now - g.start) / 1000))} 秒 · ${d.reason}`;
    if (progress >= 1 && calibration) {
      if (!fresh || !d.ready || g.counts.left < 10 || g.counts.right < 10) {
        g.start = now - 6000;
        toast("双腕还不够清晰，请把手收进画面再试");
        return;
      }
      const xs = g.samples.map((p) => p.x).sort((a, b) => a - b),
        ys = g.samples.map((p) => p.y).sort((a, b) => a - b);
      const q = (v: number[], n: number) => v[Math.floor((v.length - 1) * n)];
      const sx = q(xs, 0.97) - q(xs, 0.03),
        sy = q(ys, 0.97) - q(ys, 0.03);
      if (sx < calibration.shoulder * 0.8 || sy < calibration.shoulder * 0.6) {
        g.start = now - 4000;
        g.samples = [];
        g.counts = { left: 0, right: 0 };
        toast("请再向左右和高低伸展，让目标位于舒适范围");
        return;
      }
      calibration.reachX = Math.max(calibration.shoulder * 1.8, sx * 1.1);
      calibration.reachY = Math.max(calibration.shoulder * 1.6, sy * 1.1);
      calibration.centerY = (q(ys, 0.03) + q(ys, 0.97)) / 2;
      store("pulse-xy-calibration", calibration);
      tracker.reset(calibration);
      verified = true;
      guide = null;
      $("#guide")?.remove();
      go("home");
      toast("上半身定位完成，可以进入舞台");
      return;
    }
  }
  $("#guide-state").textContent = fresh
    ? d.ready
      ? "上半身可用，跟随提示完成动作"
      : d.reason
    : "等待新画面";
  $("#guide-detail").textContent = detail;
  $("#guide-bar").style.width = Math.min(100, progress * 100) + "%";
}

async function start() {
  if (loading) return;
  if (input === "camera" && !camera.active) {
    await enable();
    return;
  }
  if (input === "camera" && !verified) {
    view = "lab";
    render();
    beginGuide();
    return;
  }
  if (input === "camera" && !bodyReady()) {
    toast("请让双肩、双肘和双腕清晰入镜，再开始");
    return;
  }
  loading = true;
  const token = ++generation;
  try {
    const r = await fetch("/charts/xy-64.json");
    if (!r.ok) throw Error("谱面读取失败");
    const chart: Chart = await r.json();
    const errors = validateChart(chart);
    if (
      chart.ruleVersion !== XY_RULE ||
      chart.mode !== "XY" ||
      chart.notes.some((n) => !["slash", "double"].includes(n.kind))
    )
      throw Error("谱面不是当前XY规则");
    if (errors.length) throw Error(errors.join("/"));
    await music.init();
    if (token !== generation) return;
    game = new XYGame(chart);
    game.assisted = input === "keyboard";
    stage.game = game;
    stage.frame = input === "camera" ? frame : null;
    stage.clearTrails();
    stage.simulationHands = { left: null, right: null };
    stats = { processing: [], fps: [], interruptions: [], judgements: [] };
    tracker.reset(calibration);
    missingSince = { left: 0, right: 0, shoulders: 0 };
    lowSince = 0;
    aboveSince = 0;
    game.onJudge = (j) => {
      stage.flash(j);
      music.hit(j.grade);
      stats.judgements.push({
        id: j.note.id,
        grade: j.grade,
        error: j.error,
        factor: j.factor,
        action: j.action,
      });
    };
    view = "play";
    playing = true;
    paused = false;
    counting = true;
    finishing = false;
    render();
    music.play(0, 2);
  } catch (e) {
    toast("无法开始：" + String(e));
  } finally {
    loading = false;
  }
}
function pause(reason: string) {
  if (!playing || !game) return;
  music.pause();
  playing = false;
  paused = true;
  counting = false;
  game.assisted = true;
  tracker.reset(calibration);
  stats.interruptions.push(reason);
  resetMenu();
  const el = $("#pause-overlay");
  el.hidden = false;
  el.className = "modal-backdrop";
  el.innerHTML = `<div class="modal"><div class="eyebrow">STAGE PAUSED</div><h2>舞台等你回来。</h2><p>${escape(reason)}</p><p class="fineprint">继续会从安全乐句恢复，并重算回放区间；本局为辅助练习。</p><div class="modal-actions"><button class="text-btn" data-back>返回准备</button><button class="secondary" id="restart">重新开始</button><button class="primary" id="resume">继续 ▷</button></div></div>`;
  el.querySelector<HTMLElement>("[data-back]")!.onclick = () => go("home");
  $("#restart").onclick = start;
  $("#resume").onclick = async () => {
    if (input === "camera" && (!bodyReady() || camera.health.pause)) {
      toast("请先恢复上半身输入；输入检查可帮助定位问题");
      return;
    }
    try {
      await music.init();
      const from = Math.max(0, Math.floor(music.offset / 4) * 4);
      game!.rewind(from);
      tracker.reset(calibration);
      stage.clearTrails();
      music.play(from, 2);
      playing = true;
      paused = false;
      counting = true;
      missingSince = { left: 0, right: 0, shoulders: 0 };
      lowSince = 0;
      aboveSince = 0;
      el.hidden = true;
      resetMenu();
    } catch (e) {
      toast(String(e));
    }
  };
}
function bestKey() {

function finish() {
  if (!game || finishing) return;
  game.tick(65);
  playing = false;
  counting = false;
  paused = false;
  finishing = true;
  music.pause();
  tracker.reset(calibration);
  resetMenu();
  $("#pause").hidden = true;
  $("#center-message").innerHTML =
    '<strong class="complete-title">SONG<br>COMPLETE</strong><span>一分钟的光，由你点亮</span>';
  const token = generation;
  setTimeout(() => {
    if (token !== generation || !game) return;
    const bests = safe<Record<string, number>>("pulse-xy-bests", {});
    if (!game.assisted && input === "camera") {
      bests[bestKey()] = Math.max(bests[bestKey()] ?? 0, game.score);
      store("pulse-xy-bests", bests);
    }
    finishing = false;
    view = "results";
    render();
  }, 1400);
}
function dispatch(a: Action) {
  if (!playing || !game || counting || music.time < 0) return;
  const time = music.eventTime(a.time),
    prevTime = a.prevTime === undefined ? time : music.eventTime(a.prevTime);
  game.consume({ ...a, time, prevTime });
}

camera.onPose = (points, time, processing) => {
  lastPose = performance.now();
  if (guide?.step === 0) capture.add(points, time);
  if (guide?.step === 3) {
    const upper = inspectUpper(points);
    for (const hand of ["left", "right"] as const) {
      const i = hand === "left" ? 15 : 16;
      if (upper[hand]) {
        guide.samples.push(points[i]);
        guide.counts[hand]++;
      }
    }
  }
  const previous = frame;
  frame = tracker.process(points, time, processing);
  if (input === "camera") stage.frame = frame;
  const d = inspectUpper(points);
  for (const hand of ["left", "right"] as const) {
    if (d[hand]) missingSince[hand] = 0;
    else if (!missingSince[hand]) missingSince[hand] = lastPose;
  }
  if (d.shoulders) missingSince.shoulders = 0;
  else if (!missingSince.shoulders) missingSince.shoulders = lastPose;
  if (!playing || input !== "camera" || !game || counting) return;
  const at = music.eventTime(time);
  if (previous && time - previous.time > 150)
    for (const hand of ["left", "right"] as const)
      game.protectHand(music.eventTime(previous.time), at, hand);
  for (const hand of ["left", "right"] as const)
    if (!d[hand]) game.protectHand(at - 0.26, at, hand);
  for (const a of frame.actions) dispatch(a);
  stats.processing.push(processing);
  stats.fps.push(camera.fps);
};
let keyId = 0;
function keyAction(hand: "left" | "right") {
  if (!playing || !game || counting) return;
  const at = music.time;
  const n = game.chart.notes
    .filter(
      (n) =>
        !game!.results.has(n.id) &&
        (!n.hand || n.hand === hand) &&
        !game!.hits.get(n.id)?.has(hand),
    )
    .sort((a, b) => Math.abs(a.time - at) - Math.abs(b.time - at))[0];
  if (!n || Math.abs(n.time - at) > 0.4) return;
  const x = n.kind === "double" ? (hand === "left" ? 0.32 : 0.68) : n.x;
  stage.simulationHands[hand] = { x, y: n.y };
  stage.simulationAt = performance.now();
  dispatch({
    id: `key-${++keyId}`,
    kind: "hand",
    hand,
    time: performance.now(),
    prevTime: performance.now(),
    x: x + 0.1,
    y: n.y,
    prevX: x - 0.1,
    prevY: n.y,
    dx: 0.2,
    dy: 0,
    speed: 2,
    confidence: 1,
  });
}

function report() {
  const percentile = (a: number[], q: number) =>
    a.length
      ? [...a].sort((x, y) => x - y)[Math.floor((a.length - 1) * q)]
      : null;
  const data = {
    app: "Pulse Stage",
    version: VERSION,
    requirement: "1.3",
    rule: XY_RULE,
    createdAt: new Date().toISOString(),
    input,
    model: POSE_MODEL,
    userAgent: navigator.userAgent,
    calibration,
    inputHealth: camera.health,
    chart: game?.chart,
    game: game
      ? {
          score: game.score,
          accuracy: game.accuracy,
          counts: game.counts,
          maxCombo: game.maxCombo,
          assisted: game.assisted,
        }
      : null,
    metrics: {
      inferenceP95Ms: percentile(stats.processing, 0.95),
      poseFpsMedian: percentile(stats.fps, 0.5),
      endToEndLatencyMs: null,
      humanValidatedAccuracy: null,
    },
    stats,
    privacy: "no raw video or continuous landmarks",
    verification:
      "Keyboard/synthetic tests are not real-camera accuracy evidence.",
  };
  const url = URL.createObjectURL(
    new Blob([JSON.stringify(data, null, 2)], { type: "application/json" }),
  );
  const a = document.createElement("a");
  a.href = url;
  a.download = `pulse-xy-report-${Date.now()}.json`;
  a.click();
  setTimeout(() => URL.revokeObjectURL(url), 2000);
}

        el.style.top = p.y * 100 + "%";
      }
    }
  const canvas = $<HTMLCanvasElement>("#skeleton");
  if (!canvas || !preview) return;
  const r = canvas.getBoundingClientRect();
  if (!r.width || !r.height) return;
  canvas.width = r.width * devicePixelRatio;
  canvas.height = r.height * devicePixelRatio;
  const ctx = canvas.getContext("2d")!;
  ctx.scale(devicePixelRatio, devicePixelRatio);
  const videoWidth = camera.video.videoWidth || 640,
    videoHeight = camera.video.videoHeight || 480,
    s = Math.min(r.width / videoWidth, r.height / videoHeight),
    w = videoWidth * s,
    h = videoHeight * s;
  ctx.translate((r.width - w) / 2, (r.height - h) / 2);
  const points = frame?.points ?? [];
  const xy = (i: number) => ({ x: (1 - points[i].x) * w, y: points[i].y * h });
  for (const [a, b] of [
    [11, 12],
    [11, 13],
    [13, 15],
    [12, 14],
    [14, 16],
  ])
    if (points[a] && points[b]) {
      ctx.strokeStyle =
        jointStatus(points[a]) === "ready" && jointStatus(points[b]) === "ready"
          ? "#69e9ff"
          : "#dca87066";
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(xy(a).x, xy(a).y);
      ctx.lineTo(xy(b).x, xy(b).y);
      ctx.stroke();
    }
  for (const j of d.joints)
    if (points[j.id] && jointStatus(points[j.id]) !== "outside") {
      ctx.fillStyle =
        j.status === "ready"
          ? j.id === 15
            ? "#ffd27a"
            : "#69e9ff"
          : "#dca87088";
      ctx.beginPath();
      ctx.arc(xy(j.id).x, xy(j.id).y, 4, 0, Math.PI * 2);
      ctx.fill();
    }
}
function dwell(now: number) {
  const cursor = $("#hand-cursor"),
    modal = $("#guide");
  if (
    !camera.active ||
    !verified ||
    !frame?.hands[mainHand] ||
    now - frame.time > 250 ||
    playing ||
    finishing ||
    modal
  ) {
    cursor.classList.remove("visible");
    dwellState.el = null;
    dwellState.since = now;
    return;
  }
  const p = frame.hands[mainHand]!,
    x = p.x * innerWidth,
    y = p.y * innerHeight;
  cursor.classList.add("visible");
  cursor.style.left = x + "px";
  cursor.style.top = y + "px";
  const target =
    document
      .elementFromPoint(x, y)
      ?.closest<HTMLElement>("button:not(:disabled)") ?? null;
  if (!target) {
    dwellState.armed = true;
    dwellState.el = null;
    cursor.style.setProperty("--progress", "0deg");
    return;
  }
  if (!dwellState.armed) return;
  if (dwellState.el !== target) {
    dwellState.el = target;
    dwellState.since = now;
  }
  const progress = (now - dwellState.since) / 1000;
  cursor.style.setProperty("--progress", Math.min(360, progress * 360) + "deg");
  if (progress >= 1) {
    dwellState = { el: null, since: now, armed: false };
    target.click();
  }
}

function animate(now: number) {
  requestAnimationFrame(animate);
  stage.time = music.time;
  stage.draw(now);
  if (guide) guideTick(now);
  if (view === "play" && playing && game) {
    const time = music.time;
    if (counting) {
      if (time < 0 || time < music.offset) {
        const remaining = music.started - (music.ctx?.currentTime ?? 0);
        $("#center-message").innerHTML =
          `<strong>${Math.max(1, Math.ceil(remaining / 0.5))}</strong><span>音符到面，划过光点</span>`;
      } else {
        counting = false;
        $("#center-message").innerHTML = "";
        tracker.reset(calibration);
      }
    }
    if (!counting) {
      if (music.ctx && music.ctx.state !== "running")
        pause("音乐输出已暂停，请恢复音频后继续");
      if (input === "camera") {
        const health = camera.health;
        if (health.pause) pause(health.message);
        if (health.slow) {
          if (!lowSince) lowSince = now;
          if (now - lowSince > 2000) {
            if (!game.assisted) {
              stats.interruptions.push("slow-inference");
              toast("识别偏慢，本局转为辅助练习，已降低特效");
            }
            game.assisted = true;
            stage.low = true;
          }
        } else lowSince = 0;
        for (const hand of ["left", "right"] as const)
          if (missingSince[hand] && now - missingSince[hand] > 800)
            pause(
              `${hand === "left" ? "左" : "右"}手臂持续不可用，请收回取景范围`,
            );
        if (missingSince.shoulders && now - missingSince.shoulders > 500)
          pause("双肩定位暂时丢失，请正面回到画面中");
        if (frame && now - frame.time > 250)
          for (const hand of ["left", "right"] as const)
            game.protectHand(time - 0.26, time, hand);
        $("#tracking-status").textContent = game.assisted
          ? "辅助练习 · " +
            (health.slow ? health.message : (frame?.reason ?? "等待输入"))
          : `${camera.fps.toFixed(0)} FPS · ${frame?.reason ?? "等待输入"}`;
        const p = frame?.points;
        if (
          frame &&
          now - frame.time < 250 &&
          p &&
          jointStatus(p[15]) === "ready" &&
          jointStatus(p[16]) === "ready" &&
          jointStatus(p[0]) === "ready" &&
          p[15].y < p[0].y - 0.02 &&
          p[16].y < p[0].y - 0.02 &&
          Math.abs(p[15].x - p[16].x) < 0.2
        ) {
          if (!aboveSince) aboveSince = now;
          if (now - aboveSince > 1500) pause("举手暂停");
        } else aboveSince = 0;
      }
      if (playing) {
        game.tick(time);
        if (time >= 64.05) finish();
      }
    }
    $("#score").textContent = String(game.score).padStart(7, "0");
    $("#combo").textContent = String(game.combo);
    $("#time").textContent =
      `${Math.floor(Math.max(0, time) / 60)}:${String(Math.floor(Math.max(0, time)) % 60).padStart(2, "0")}`;
    $("#progress").style.width =
      Math.min(100, (Math.max(0, time) / 64) * 100) + "%";
  }
  if (now - lastPaint > 100) {
    lastPaint = now;
    paintInput();
    device();
    $("#fps").textContent =
      `${stage.fps.toFixed(0)} RENDER FPS${camera.active ? " / " + camera.fps.toFixed(0) + " POSE FPS" : ""}`;
  }
  dwell(now);
}
